-- 1. PATIENTS TABLE
CREATE TABLE Patients (
    PatientID           VARCHAR(50) PRIMARY KEY,
    FirstName           NVARCHAR(100) NOT NULL,
    LastName            NVARCHAR(100) NOT NULL,
    DateOfBirth         DATE NOT NULL,
    Gender              VARCHAR(10) CHECK (Gender IN ('Male', 'Female', 'Other')),
    AddressLine1        NVARCHAR(255),
    City                NVARCHAR(100),
    Province            NVARCHAR(50),
    PostalCode          NVARCHAR(20),
    PhoneNumber         NVARCHAR(20),
    Email               NVARCHAR(255),
    EmployerName        NVARCHAR(255),
    IndustryType        NVARCHAR(100),
    InjuryDate          DATE,
    ClaimStartDate      DATE,
    ClaimEndDate        DATE NULL,
    ReturnToWorkStatus  VARCHAR(50) CHECK (ReturnToWorkStatus IN ('Pending', 'In Progress', 'Completed', 'Unknown')),
    CreatedAt           DATETIME2 DEFAULT SYSUTCDATETIME()
);

-- 2. PROVIDERS TABLE
CREATE TABLE Providers (
    ProviderID          VARCHAR(50) PRIMARY KEY,
    ProviderName        NVARCHAR(255) NOT NULL,
    ProviderType        VARCHAR(100) CHECK (ProviderType IN ('Physician', 'Physiotherapist', 'Chiropractor', 'Hospital', 'Clinic', 'Other')),
    Specialization      NVARCHAR(100),
    LicenseNumber       VARCHAR(50),
    AddressLine1        NVARCHAR(255),
    City                NVARCHAR(100),
    Province            NVARCHAR(50),
    PostalCode          NVARCHAR(20),
    PhoneNumber         NVARCHAR(20),
    Email               NVARCHAR(255),
    IsActive            BIT DEFAULT 1,
    CreatedAt           DATETIME2 DEFAULT SYSUTCDATETIME()
);

-- 3. INVOICES TABLE
CREATE TABLE Invoices (
    InvoiceID           VARCHAR(50) PRIMARY KEY,
    PatientID           VARCHAR(50) NOT NULL,
    ProviderID          VARCHAR(50) NOT NULL,
    ProcedureCode       VARCHAR(20) NOT NULL,
    ProcedureDescription NVARCHAR(500),
    BilledAmount        DECIMAL(10,2) NOT NULL,
    InvoiceDate         DATE NOT NULL,
    ServiceDate         DATE NOT NULL,
    Status              VARCHAR(50) CHECK (Status IN ('Pending Review', 'Approved', 'Denied', 'Escalated', 'Under Review')) DEFAULT 'Pending Review',
    ReviewedBy          VARCHAR(100) NULL,
    ReviewDate          DATETIME2 NULL,
    AIConfidenceScore   FLOAT NULL,
    DecisionReasoning   NVARCHAR(MAX) NULL,
    RejectionReason     NVARCHAR(500) NULL,
    CreatedAt           DATETIME2 DEFAULT SYSUTCDATETIME(),
    FOREIGN KEY (PatientID) REFERENCES Patients(PatientID),
    FOREIGN KEY (ProviderID) REFERENCES Providers(ProviderID)
);

-- 4. POLICIES TABLE
CREATE TABLE Policies (
    PolicyID            VARCHAR(50) PRIMARY KEY,
    PolicyName          NVARCHAR(255) NOT NULL,
    PolicyDescription   NVARCHAR(MAX),
    ApplicableProcedures NVARCHAR(MAX), -- Comma-separated procedure codes or 'All'
    IndustryType        NVARCHAR(100), -- Specific industry or 'All'
    MaxCoverageAmount   DECIMAL(10,2) NULL,
    RequiredDocumentation NVARCHAR(MAX),
    EffectiveDate       DATE NOT NULL,
    ExpiryDate          DATE NULL,
    IsActive            BIT DEFAULT 1,
    CreatedAt           DATETIME2 DEFAULT SYSUTCDATETIME()
);

-- 5. COMMUNICATIONS TABLE
CREATE TABLE Communications (
    CommunicationID     VARCHAR(50) PRIMARY KEY,
    PatientID           VARCHAR(50) NOT NULL,
    CommunicationType   VARCHAR(50) CHECK (CommunicationType IN ('Email', 'Phone', 'Letter', 'Fax', 'Portal')),
    Subject             NVARCHAR(255),
    MessageBody         NVARCHAR(MAX),
    SentDate            DATETIME2 DEFAULT SYSUTCDATETIME(),
    SentBy              NVARCHAR(100), -- Staff member or system
    ReceivedFrom        NVARCHAR(100), -- Patient, Provider, or Other
    AttachmentCount     INT DEFAULT 0,
    CreatedAt           DATETIME2 DEFAULT SYSUTCDATETIME(),
    FOREIGN KEY (PatientID) REFERENCES Patients(PatientID)
);

-- 6. PROCEDURE_CATALOG TABLE
CREATE TABLE ProcedureCatalog (
    ProcedureCode       VARCHAR(20) PRIMARY KEY,
    ProcedureName       NVARCHAR(255) NOT NULL,
    Category            VARCHAR(100),
    StandardCost        DECIMAL(10,2),
    TypicalDuration     INT, -- in minutes
    RequiresPreAuth     BIT DEFAULT 0,
    IsActive            BIT DEFAULT 1
);

-- 7. AUDIT_LOG TABLE
CREATE TABLE AuditLog (
    AuditID             BIGINT IDENTITY(1,1) PRIMARY KEY,
    InvoiceID           VARCHAR(50),
    Action              VARCHAR(100) NOT NULL,
    PerformedBy         VARCHAR(100) NOT NULL,
    OldValue            NVARCHAR(MAX),
    NewValue            NVARCHAR(MAX),
    Timestamp           DATETIME2 DEFAULT SYSUTCDATETIME(),
    FOREIGN KEY (InvoiceID) REFERENCES Invoices(InvoiceID)
);
